### 5. Firestore Security Rules

**File: firestore.rules**

```firestore
rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // OTP Verifications Collection
    match /otpVerifications/{otpDoc=**} {
      // Allow Cloud Functions to manage OTPs
      allow read, write: if request.auth != null;

      // Restrict to authenticated users
      allow read: if request.auth.uid != null;
      allow write: if request.auth.uid != null;
    }

    // Users Collection
    match /users/{userId} {
      // Users can read/write their own document
      allow read, write: if request.auth.uid == userId;

      // Admins can read all users
      allow read: if get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';

      // Allow Cloud Functions to read users
      allow read: if request.auth != null;
    }

    // Logs Collection (for audit)
    match /logs/{logDoc=**} {
      allow read, write: if request.auth.uid != null;
    }

    // Default deny all other access
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
```

### 8. Development Setup Script

**File: scripts/setup-dev-env.sh**

```bash
#!/bin/bash

# Setup development environment for Semaphore SMS OTP

echo "🚀 Setting up Semaphore SMS OTP development environment..."

# Check Node version
NODE_VERSION=$(node -v)
echo "✓ Node version: $NODE_VERSION"

# Check Firebase CLI
if ! command -v firebase &> /dev/null; then
    echo "❌ Firebase CLI not found. Installing..."
    npm install -g firebase-tools
fi

# Navigate to functions directory
cd functions

# Install dependencies
echo "📦 Installing dependencies..."
npm install

# Create .env.local if it doesn't exist
if [ ! -f .env.local ]; then
    echo "📝 Creating .env.local..."
    cat > .env.local << EOF
SEMAPHORE_API_KEY=sem_YOUR_TEST_API_KEY_HERE
NODE_ENV=development
EOF
    echo "⚠️  Please update .env.local with your Semaphore API key"
fi

# Check if dotenv is installed
if grep -q "dotenv" package.json; then
    echo "✓ dotenv is installed"
else
    echo "📦 Installing dotenv..."
    npm install dotenv
fi

echo "✅ Development environment setup complete!"
echo ""
echo "Next steps:"
echo "1. Update functions/.env.local with your Semaphore API key"
echo "2. Run: firebase emulators:start --only functions"
echo "3. Test with: firebase functions:shell"
```

**Make executable:**

```bash
chmod +x scripts/setup-dev-env.sh
./scripts/setup-dev-env.sh
```

### 9. Production Deployment Script

**File: scripts/deploy-semaphore.sh**

```bash
#!/bin/bash

# Deploy Semaphore SMS OTP to production

echo "🚀 Deploying Semaphore SMS OTP..."

# Check if API key is set
if [ -z "$SEMAPHORE_API_KEY" ]; then
    echo "❌ SEMAPHORE_API_KEY environment variable not set"
    exit 1
fi

# Navigate to functions directory
cd functions

# Install dependencies
echo "📦 Installing dependencies..."
npm install

# Run linter if available
if [ -f ".eslintrc" ]; then
    echo "🔍 Running linter..."
    npm run lint
fi

# Deploy Cloud Functions
echo "🚀 Deploying Cloud Functions..."
firebase deploy --only functions

if [ $? -eq 0 ]; then
    echo "✅ Deployment successful!"
    echo ""
    echo "Deployed functions:"
    firebase functions:list
else
    echo "❌ Deployment failed"
    exit 1
fi

# View logs
echo ""
echo "📋 Recent logs:"
firebase functions:log --limit=10
```

**Make executable:**

```bash
chmod +x scripts/deploy-semaphore.sh
SEMAPHORE_API_KEY="sem_YOUR_KEY" ./scripts/deploy-semaphore.sh
```

### 10. Testing Template

**File: functions/test-semaphore.js**

```javascript
/**
 * Test script for Semaphore SMS OTP functions
 * Run with: node test-semaphore.js
 */

const admin = require("firebase-admin");
const serviceAccount = require("./serviceAccountKey.json");

// Initialize Firebase Admin
admin.initializeApp({
  credential: admin.credential.cert(serviceAccount),
  projectId: "YOUR_PROJECT_ID",
});

const db = admin.firestore();

// Test OTP generation
function testGenerateOTP() {
  console.log("\n🧪 Testing OTP Generation...");
  const otp = Math.floor(100000 + Math.random() * 900000).toString();
  console.log("Generated OTP:", otp);
  console.assert(otp.length === 6, "OTP should be 6 digits");
  console.log("✅ OTP generation test passed");
}

// Test phone formatting
function testPhoneFormatting() {
  console.log("\n🧪 Testing Phone Number Formatting...");

  function formatPhoneNumber(phone) {
    if (!phone) return null;
    let cleaned = phone.replace(/[\s\-\(\)]/g, "");
    if (cleaned.startsWith("+63")) {
      return "0" + cleaned.substring(3);
    }
    if (cleaned.startsWith("09")) {
      return cleaned;
    }
    if (cleaned.startsWith("63")) {
      return "0" + cleaned.substring(2);
    }
    return cleaned;
  }

  const testCases = [
    { input: "+639171234567", expected: "09171234567" },
    { input: "+63 917 123 4567", expected: "09171234567" },
    { input: "09171234567", expected: "09171234567" },
    { input: "639171234567", expected: "09171234567" },
  ];

  testCases.forEach((testCase) => {
    const result = formatPhoneNumber(testCase.input);
    console.assert(
      result === testCase.expected,
      `Format failed for ${testCase.input}`,
    );
    console.log(`✓ ${testCase.input} → ${result}`);
  });

  console.log("✅ Phone formatting tests passed");
}

// Test Firestore OTP storage
async function testFirestoreStorage() {
  console.log("\n🧪 Testing Firestore OTP Storage...");

  try {
    const testOTPDoc = {
      phone: "+639171234567",
      otp: "123456",
      createdAt: admin.firestore.Timestamp.now(),
      expiresAt: admin.firestore.Timestamp.now(),
      attempts: 0,
      verified: false,
    };

    // Test create
    await db.collection("otpVerifications").doc("test_otp").set(testOTPDoc);
    console.log("✓ OTP created in Firestore");

    // Test read
    const doc = await db.collection("otpVerifications").doc("test_otp").get();
    console.assert(doc.exists, "Document should exist");
    console.log("✓ OTP retrieved from Firestore");

    // Test update
    await db.collection("otpVerifications").doc("test_otp").update({
      attempts: 1,
    });
    console.log("✓ OTP updated in Firestore");

    // Test delete
    await db.collection("otpVerifications").doc("test_otp").delete();
    console.log("✓ OTP deleted from Firestore");

    console.log("✅ Firestore storage tests passed");
  } catch (error) {
    console.error("❌ Firestore test failed:", error.message);
  }
}

// Test OTP expiry
function testOTPExpiry() {
  console.log("\n🧪 Testing OTP Expiry Logic...");

  function isOTPExpired(createdAt) {
    if (!createdAt) return true;
    const now = new Date();
    const otpAge = (now - createdAt.toDate()) / 1000 / 60;
    return otpAge > 10; // 10 minutes
  }

  // Test fresh OTP
  const freshTime = admin.firestore.Timestamp.now();
  console.assert(!isOTPExpired(freshTime), "Fresh OTP should not be expired");
  console.log("✓ Fresh OTP not expired");

  // Test expired OTP (simulated)
  const expiredTime = new Date(Date.now() - 15 * 60 * 1000);
  const expiredTimestamp = admin.firestore.Timestamp.fromDate(expiredTime);
  console.assert(isOTPExpired(expiredTimestamp), "Old OTP should be expired");
  console.log("✓ Old OTP expired");

  console.log("✅ OTP expiry tests passed");
}

// Run all tests
async function runAllTests() {
  console.log("🚀 Running Semaphore SMS OTP Tests...\n");

  testGenerateOTP();
  testPhoneFormatting();
  testOTPExpiry();
  await testFirestoreStorage();

  console.log("\n✅ All tests completed!");
  process.exit(0);
}

runAllTests().catch((error) => {
  console.error("❌ Test suite failed:", error);
  process.exit(1);
});
```

```bash

node test-semaphore.js
```
